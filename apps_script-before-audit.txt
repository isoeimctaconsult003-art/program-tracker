function schConflict_(b, all) {
  const day = all.filter(x => x.date === b.date), mine = day.filter(x => x.consultant === b.consultant);
  if (mine.some(x => x.shift === b.shift)) return b.consultant + " already has a " + b.shift.toLowerCase() + " booking on this date.";
  if (mine.length >= 2) return b.consultant + " already has 2 schedules on this date.";
  const bp = b.programs || [];
  const clash = day.find(x => x.school === b.school && x.shift === b.shift && x.consultant !== b.consultant && x.programs.some(p => bp.indexOf(p) >= 0));
  if (clash) return "At " + b.school + " in the " + b.shift.toLowerCase() + ", " + clash.consultant + " already has " + clash.programs.filter(p => bp.indexOf(p) >= 0).join(", ") + ". Two consultants can share a school and shift only with different programs.";
  return "";
}

function makeInvitation_(b) {
  let stage = "copy the template", file = null;
  try {
    file = DriveApp.getFileById(TEMPLATE_DOC_ID).makeCopy("Invitation - " + b.school + " - " + b.date);
    stage = "move to folder";
    if (INVITE_FOLDER_ID) file.moveTo(DriveApp.getFolderById(INVITE_FOLDER_ID));
    stage = "open the copy";
    const doc = DocumentApp.openById(file.getId()), body = doc.getBody();
    stage = "school line";
    const ps = body.getParagraphs();
    for (let i = 0; i < ps.length; i++) {
      if (/^Dear Valued/i.test(ps[i].getText().trim())) {
        for (let j = i + 1; j < ps.length; j++) {
          if (ps[j].getText().trim()) { ps[j].replaceText(".+", esc_(b.school)); break; }
        }
        break;
      }
    }
    const val = {
      "program title": progTitle_(b),
      "expected participants": (b.participation || []).join(", "),
      "venue": b.school,
      "date": b.bs || fmtDate_(b.date),
      "time": b.time,
      "presenter": ((TITLES[b.consultant] || "") + " " + b.consultant).trim(),
      "required resources": b.resources
    };
    const tables = body.getTables();
    for (let ti = 0; ti < tables.length; ti++) {
      const t = tables[ti];
      for (let r = 0; r < t.getNumRows(); r++) {
        const row = t.getRow(r);
        if (row.getNumCells() < 2) continue;
        const k = row.getCell(0).getText().trim().toLowerCase();
        if (k in val) { stage = "table row '" + k + "'"; setCell_(row.getCell(1), val[k]); }
      }
    }
    stage = "save";
    doc.saveAndClose();
    stage = "share";
    if (SHARE_LINK) file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return file.getUrl();
  } catch (err) {
    try { if (file) file.setTrashed(true); } catch (x) {}
    throw new Error("Invitation step '" + stage + "' failed: " + err);
  }
}

function hex_(bytes) { return bytes.map(b => ("0" + (b & 0xff).toString(16)).slice(-2)).join(""); }
function hashPw_(salt, pw) {
  let h = salt + "|" + pw;
  for (let i = 0; i < 1200; i++) h = hex_(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, h + salt));
  return h;
}
function secret_() {
  const p = PropertiesService.getScriptProperties();
  let s = p.getProperty("SECRET");
  if (!s) { s = Utilities.getUuid() + Utilities.getUuid(); p.setProperty("SECRET", s); }
  return s;
}
function sign_(payload) { return Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(payload, secret_())); }
function makeToken_(u) {
  const payload = Utilities.base64EncodeWebSafe(JSON.stringify({e: u.email, x: Date.now() + SESSION_DAYS * 864e5}));
  return payload + "." + sign_(payload);
}

function verify_(tok) {
  if (!tok) return null;
  const parts = String(tok).split(".");
  if (parts.length !== 2 || sign_(parts[0]) !== parts[1]) return null;
  let p;
  try { p = JSON.parse(Utilities.newBlob(Utilities.base64DecodeWebSafe(parts[0])).getDataAsString()); } catch (e) { return null; }
  if (!p || !p.x || p.x < Date.now()) return null;
  const u = findUser_(p.e);
  if (!u || u.status !== "Active") return null;
  return {email: u.email, name: u.name, role: u.role, username: u.username, branch: userBranch_(u)};
}

  const u = Utilities.getUuid().replace(/-/g, "") + Utilities.getUuid().replace(/-/g, "");

function doLogin_(d) {
  const email = String(d.email || "").trim().toLowerCase(), cache = CacheService.getScriptCache(), ck = "fail:" + email;
  if (Number(cache.get(ck) || 0) >= 5) return {ok: false, error: "Too many attempts. Please try again in 10 minutes."};
  const u = findUser_(email);
  if (!u || u.status !== "Active" || hashPw_(u.salt, String(d.password || "")) !== u.hash) {
    cache.put(ck, String(Number(cache.get(ck) || 0) + 1), 600);
    return {ok: false, error: "Wrong email or password."};
  }
  cache.remove(ck);
  return {ok: true, token: makeToken_(u), user: {email: u.email, name: u.name, role: u.role, username: u.username, branch: userBranch_(u)}};
}

function doReset_(d) {
  const email = String(d.email || "").trim().toLowerCase(), cache = CacheService.getScriptCache(), ck = "rs:" + email, pw = String(d.password || "");
  if (Number(cache.get(ck) || 0) >= 5) return {ok: false, error: "Too many attempts. Please ask for a new code later."};
  if (pw.length < 8) return {ok: false, error: "The new password must be at least 8 characters."};
  const u = findUser_(email);
  if (!u || u.status !== "Active" || !u.resetHash || Date.now() > u.resetExp || hashPw_(u.salt, String(d.code || "").trim()) !== u.resetHash) {
    cache.put(ck, String(Number(cache.get(ck) || 0) + 1), 600);
    return {ok: false, error: "The code is wrong or has expired."};
  }
  setPw_(u, pw);
  usersSheet_().getRange(u.row, 8, 1, 2).setValues([["", ""]]);
  cache.remove(ck);
  return {ok: true};
}
function doChangePw_(me, d) {
  const u = findUser_(me.email), pw = String(d.password || "");
  if (!u || hashPw_(u.salt, String(d.old || "")) !== u.hash) return {ok: false, error: "The current password is wrong."};
  if (pw.length < 8) return {ok: false, error: "The new password must be at least 8 characters."};
  setPw_(u, pw);
  return {ok: true};
}
function adminAction_(me, d) {
  const isSuper = me.role === "admin", myBranch = userBranch_(me);
  const rowUser = r => ({email: String(r[0]), name: String(r[1]), role: String(r[2]), status: String(r[5]), username: String(r[9] || ""), branch: String(r[10] || "") || branchOfName_(r[1])});

    const rows = usersSheet_().getDataRange().getValues().slice(1).filter(r => r[0]).map(rowUser).filter(manage);
    return {ok: true, users: rows};
  }
  if (d.action === "addUser") {
    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) return {ok: false, error: "Enter a valid email address."};
    if (findUser_(email)) return {ok: false, error: "This email already has an account."};

    if (!/^[a-z0-9._-]{3,30}$/.test(username)) return {ok: false, error: "The username must be 3 to 30 letters, numbers, dots, dashes or underscores."};
    if (findUser_(username)) return {ok: false, error: "This username is already taken."};
    let role = ["admin", "branch_admin", "consultant"].indexOf(d.role) >= 0 ? d.role : "consultant", branch = BRANCHES[d.branch] ? d.branch : "";
    if (!isSuper) { role = "consultant"; branch = myBranch; }
    if (role === "branch_admin" && !branch) return {ok: false, error: "Choose the branch for a branch administrator."};
    const name = String(d.name || email).trim(), pw = tempPw_();
    addUser_(email, name, role, pw, username, branch);

    try { MailApp.sendEmail(email, "Your Third Eye login", "Hello " + name + ",\n\nYour account for Third Eye by EIMCTA is ready.\nWebsite: " + DASH_URL + "\nUsername: " + username + " (or your email)\nTemporary password: " + pw + "\n\nPlease sign in and change your password (use the Password button at the top).\n"); } catch (err) { mailed = false; }

  }
  const u = findUser_(email);

  if (d.action === "setUserStatus") {

    usersSheet_().getRange(u.row, 6).setValue(d.status === "Disabled" ? "Disabled" : "Active");

    return {ok: true, temp: pw, mailed: mailed};
  }
  return {ok: false, error: "Unknown action."};
}

function doGet(e) {
  const me = verify_(e && e.parameter && e.parameter.tok);
  if (!me) return json_({__auth: false, __version: "travel-v3"});

  const ss = SpreadsheetApp.openById(SS_ID);
  const out = {};
  ss.getSheets().forEach(sh => {
    const n = sh.getName();
    if (n === TARGET_TAB || n === SCH_TAB || n === KV_TAB || n === USERS_TAB || n === TRACK_TAB) return;
    const v = sh.getDataRange().getDisplayValues();
    const h = v.findIndex(r => String(r[0]).trim() === 'S.N');
    if (h < 0) return;
    out[n.trim()] = { headers: v[h], rows: v.slice(h + 1).filter(r => String(r[1]).trim()) };
  });
  out.__version = "travel-v3";
  out.__user = me;
  out.__targets = readTargets_();
  out.__schedules = readSchedules_().filter(s => visible_(me, s.consultant));
  const kv = readKV_();
  Object.keys(kv).forEach(k => {
    if (k.indexOf("trip:") === 0 && !(kv[k] && visible_(me, kv[k].c))) delete kv[k];
    else if (k.indexOf("loc:home:") === 0 && !visible_(me, k.slice(9))) delete kv[k];
  });
  out.__kv = kv;
  out.__tracks = readTracks_().filter(t => visible_(me, t.name));
  return json_(out);
}
function doPost(e) {
  const lock = LockService.getScriptLock();
  let locked = false;
  try {
    const d = JSON.parse(e.postData.contents);
    if (d.action === "login") return json_(doLogin_(d));
    if (d.action === "forgot") return json_(doForgot_(d));
    const me = verify_(d.tok);
    if (d.action !== "reset" && !me) return json_({ok: false, error: "auth"});
    locked = lock.tryLock(28000);
    if (!locked) return json_({ok: false, error: "The server is busy. Please click Save again in a few seconds."});
    if (d.action === "reset") return json_(doReset_(d));
    if (d.action === "changePassword") return json_(doChangePw_(me, d));
    if (["listUsers", "addUser", "setUserStatus", "resetUserPassword"].indexOf(d.action) >= 0) {
      if (me.role !== "admin" && me.role !== "branch_admin") return json_({ok: false, error: "Only an administrator can do this."});
      return json_(adminAction_(me, d));
    }
    if (d.action === "setKV") {
      const k = String(d.k || "");
      if (!k) return json_({ok: false, error: "Missing key."});
      if ((k === "settings" || k.indexOf("loc:home:") === 0 || k.indexOf("loc:office:") === 0) && me.role !== "admin") return json_({ok: false, error: "Only an administrator can change this."});
      if (k.indexOf("trip:") === 0) {
        if (d.value === null) { if (me.role !== "admin") return json_({ok: false, error: "Only an administrator can delete a trip."}); }
        else {
          if (!visible_(me, d.value.c)) return json_({ok: false, error: "You can only save your own trips."});
          if ((d.value.status === "Approved" || d.value.status === "Paid") && me.role === "consultant") return json_({ok: false, error: "Only an administrator or branch administrator can approve trips."});
        }
      }
      setKV_(k, d.value);
      return json_({ok: true});
    }
    if (["deleteSchedule", "setScheduleStatus", "makeInvitation"].indexOf(d.action) >= 0 && !schAllowed_(me, d.id)) return json_({ok: false, error: "You can only change your own schedules."});
    if (d.action === "saveSchedule" && !(d.row && visible_(me, d.row.consultant))) return json_({ok: false, error: "You can only book schedules for yourself."});
    if (d.action === "saveTrack") {
      const r = d.row || {};
      if (!r.id) return json_({ok: false, error: "Missing trip id."});
      const sh = tracksSheet_();
      if (findRow_(sh, String(r.id))) return json_({ok: true, id: r.id, dup: true});
      const mode = ["bike", "car", "bus"].indexOf(r.mode) >= 0 ? r.mode : "", fare = mode === "bus" && r.fare !== null && r.fare !== undefined && !isNaN(Number(r.fare)) ? Number(r.fare) : "";
      sh.getRange(sh.getLastRow() + 1, 1, 1, 24).setNumberFormat("@").setValues([[String(r.id), r.date, me.email, me.name, userBranch_(me), r.start, r.end, r.durMin, r.km, r.avgKmh, r.maxKmh,
        JSON.stringify(r.from || {}), JSON.stringify(r.dests || []), JSON.stringify(r.stops || []), String(r.route || "").slice(0, 45000), new Date().toISOString(),
        r.startLocal || "", r.endLocal || "", r.startPlace || "", r.endPlace || "", String(r.stopsText || "").slice(0, 45000), JSON.stringify(r.endPt || {}), mode, fare]]);
      return json_({ok: true, id: r.id});
    }
    if (d.action === "deleteTrack") {
      if (me.role !== "admin") return json_({ok: false, error: "Only an administrator can delete a tracked trip."});
      const sh = tracksSheet_(), r = findRow_(sh, String(d.id));
      if (r) sh.deleteRow(r);
      return json_({ok: true});
    }
    if (d.action === "deleteSchedule") {
      const sh = schSheet_(), r = findRow_(sh, d.id);
      if (r) sh.deleteRow(r);
      return json_({ok: true});
    }
    if (d.action === "setScheduleStatus") {
      if (me.role !== "admin" && me.role !== "branch_admin") return json_({ok: false, error: "Only an administrator can mark a schedule as done."});
      const sh = schSheet_(), r = findRow_(sh, d.id);
      if (!r) return json_({ok: false, error: "Schedule not found."});
      sh.getRange(r, 13).setNumberFormat("@").setValue(d.status === "Done" ? "Done" : "Booked");
      return json_({ok: true});
    }
    if (d.action === "makeInvitation") {
      const sh = schSheet_(), r = findRow_(sh, d.id), b = readSchedules_().find(x => x.id === d.id);
      if (!r || !b) return json_({ok: false, error: "Schedule not found."});
      let url = "";
      try {
        url = makeInvitation_(b);
      } catch (err) {
        return json_({ok: false, error: "Could not create the invitation. Run authorizeAll in Apps Script once, then redeploy. (" + err + ")"});
      }
      sh.getRange(r, 14).setNumberFormat("@").setValue(url);
      return json_({ok: true, invite: url});
    }
    if (d.action === "saveSchedule") {
      const b = d.row, msg = schConflict_(b, readSchedules_());
      if (msg) return json_({ok: false, error: msg});
      const sh = schSheet_(), id = Utilities.getUuid(), row = sh.getLastRow() + 1;
      b.id = id;
      sh.getRange(row, 1, 1, 15).setNumberFormat("@").setValues([[id, b.date, b.shift, b.time, b.school, b.consultant,
        (b.programs || []).join(SEP), (b.subs || []).join(SEP), (b.services || []).join(SEP), b.resources || "",
        (b.participation || []).join(SEP), new Date().toISOString(), "Booked", "", b.bs || ""]]);
      let url = "", inviteError = "";
      try {
        url = makeInvitation_(b);
        sh.getRange(row, 14).setValue(url);
      } catch (err) {
        inviteError = "Run authorizeAll in Apps Script once, then redeploy. (" + err + ")";
      }
      return json_({ok: true, id: id, invite: url, invite_error: inviteError});
    }
    const sh = targetSheet_();
    if (d.action === "delete") {
      const r = findRow_(sh, d.id);
      if (r) sh.deleteRow(r);
      return json_({ok: true});
    }
    const t = d.row, id = t.id || Utilities.getUuid();
    const r = t.id ? findRow_(sh, t.id) : 0;
    sh.getRange(r || sh.getLastRow() + 1, 1, 1, 6).setNumberFormat("@")
      .setValues([[id, t.school, t.program, t.discussion, t.open, t.close]]);
    return json_({ok: true, id: id});
  } catch (err) {
    return json_({ok: false, error: "Could not save: " + err});
  } finally {
    if (locked) lock.releaseLock();
  }
}

function checkClosingDates() {
  const today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  const due = readTargets_().filter(t => t.close === today);
  if (!due.length) return;
  const lines = due.map(t => "• " + t.school + ": " + t.program + (t.discussion ? " (" + t.discussion + ")" : ""));
  MailApp.sendEmail(NOTIFY_EMAIL,
    "This school has a closing date today: " + due.map(t => t.school).join(", "),
    "Closing date TODAY (" + today + "):\n\n" + lines.join("\n") + "\n\nThird Eye by EIMCTA");
}
function setupDailyReminder() {
  ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === "checkClosingDates")
    .forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger("checkClosingDates").timeBased().everyDays(1).atHour(8).create();
}
